```sql
SELECT p.id, p.posting_number, p.posting_lozon_id, p.warehouse_id, p.address_enc::BYTEA as address_enc, p.time_slot_start_at, p.time_slot_end_at, p.lat_enc::BYTEA as lat_enc, p.lng_enc::BYTEA as lng_enc, p.delivery_type, delivery_variant_id, p.courier_zone_name_enc::BYTEA as courier_zone_name_enc, p.article_box_tare_id, COUNT(p.id) OVER () AS total_points, p.delivery_name_enc, p.delivery_phone_enc, p.state, p.created_at, p.united_delivery_type, p.labels, p.scan_it 
FROM point AS p 
LEFT JOIN route_point_v2 AS rp on (rp.point_id = p.id) 
LEFT JOIN route AS r on (r.id = rp.route_id AND r.deleted_at IS NULL) 
WHERE ((r.route_lozon_id = $1 OR r.route_lozon_id IS NULL) 
AND p.deleted_at IS NULL) 
AND p.warehouse_id = $2 
AND p.time_slot_start_at >= $3 
AND p.time_slot_end_at <= $4 
AND p.time_slot_start_at >= $5 
GROUP BY p.id 
HAVING COUNT(r.id) FILTER (WHERE r.id IS NOT NULL) = 0 
LIMIT 1500 OFFSET 0
```

```sql
SELECT p.id, p.posting_number, p.posting_lozon_id, p.warehouse_id, p.address_enc::BYTEA as address_enc, p.time_slot_start_at, p.time_slot_end_at, p.lat_enc::BYTEA as lat_enc, p.lng_enc::BYTEA as lng_enc, p.delivery_type, delivery_variant_id, p.courier_zone_name_enc::BYTEA as courier_zone_name_enc, p.article_box_tare_id, COUNT(p.id) OVER () AS total_points, p.delivery_name_enc, p.delivery_phone_enc, p.state, p.created_at, p.united_delivery_type, p.labels, p.scan_it 
FROM point AS p 
INNER JOIN route_point_v2 AS rp on (rp.point_id = p.id) 
WHERE (p.state NOT IN ($1, $2, $3) AND p.deleted_at IS NULL) 
AND p.warehouse_id = $4 
AND p.time_slot_start_at >= $5 
AND p.time_slot_end_at <= $6 
AND p.time_slot_start_at >= $7 
LIMIT 1500 OFFSET 0
```
