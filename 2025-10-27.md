func Test_process(t *testing.T) {  
    t.Parallel()  
  
    now := time.Now()  
  
    type fields struct {  
       etcdClient     etcd.Client  
       appConfig      scratchconfig.RealtimeConfigClient  
       appSecret      scratchconfig.RealtimeConfigClient  
       tickPeriod     time.Duration  
       configs        []ProcessorConfig  
       repo           func(ctrl *gomock.Controller) Repository  
       policy         RetryPolicy  
       metricsManager func(ctrl *gomock.Controller) MetricsManager  
    }  
  
    tests := []struct {  
       name   string  
       fields fields  
    }{  
       {  
          name: "success",  
          fields: fields{  
             etcdClient: nil,  
             appConfig:  nil,  
             appSecret:  nil,  
             tickPeriod: time.Second,  
             configs: []ProcessorConfig{  
                {  
                   Topic:   "test-topic1",  
                   Handler: expresspostingsupdates.NewHandler(postingService),  
                },  
             },  
             repo: func(ctrl *gomock.Controller) Repository {  
                m := NewMockRepository(ctrl)  
  
                return m  
             },  
             policy: NewRetryPolicy(0, 0, 0, true, nil),  
             metricsManager: func(ctrl *gomock.Controller) MetricsManager {  
                m := NewMockMetricsManager(ctrl)  
  
                return m  
             },  
          },  
       },  
    }  
  
    for _, tt := range tests {  
       t.Run(tt.name, func(t *testing.T) {  
          t.Parallel()  
  
          ctrl := gomock.NewController(t)  
  
          s := NewRetryService(  
             tt.fields.etcdClient,  
             tt.fields.appConfig,  
             tt.fields.appSecret,  
             tt.fields.tickPeriod,  
             tt.fields.configs,  
             tt.fields.repo(ctrl),  
             tt.fields.policy,  
             tt.fields.metricsManager(ctrl),  
          )  
  
          err := s.process(context.Background(), now)  
  
          require.NoError(t, err)  
       })  
    }  
}
