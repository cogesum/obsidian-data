```sql
WITH unique_keys AS (
    SELECT DISTINCT key
    FROM retry_queue
    WHERE topic = $1
      AND (next_retry_at IS NULL OR next_retry_at <= NOW())
    ORDER BY next_retry_at
    LIMIT $2
)

SELECT rq.*
FROM retry_queue AS rq
JOIN unique_keys AS uk ON uk.key = rq.key
WHERE rq.topic = $1
  AND (rq.next_retry_at IS NULL OR rq.next_retry_at <= NOW())
ORDER BY rq.key ASC, rq.id ASC
FOR UPDATE SKIP LOCKED;
```
